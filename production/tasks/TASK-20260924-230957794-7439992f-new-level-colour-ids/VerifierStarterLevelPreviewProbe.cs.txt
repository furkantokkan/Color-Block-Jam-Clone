using System;
using System.Reflection;
using ColorBlockJam.Editor.BoardKit;
using ColorBlockJam.Editor.LevelPreview;
using ColorBlockJam.Environment;
using ColorBlockJam.Presentation.Board;
using ColorBlockJam.Runtime.Board;
using ColorBlockJam.Runtime.Layout;
using ColorBlockJam.Runtime.Levels;
using NUnit.Framework;
using UnityEditor;
using UnityEngine;
using UnityEngine.SceneManagement;

namespace ColorBlockJam.Tests.LevelPreview
{
    /// <summary>
    /// Verifier-only probe (isolated copy, never in the repo): the Create Level seed shows in the
    /// preview host with no error, and the old lowercase seed is rejected as an unknown colour.
    /// </summary>
    public sealed class VerifierStarterLevelPreviewProbe
    {
        private const string k_prefabPath = "Assets/ColorBlockJam/Prefabs/Environment/Environment.prefab";
        private const string k_databasePath = "Assets/ColorBlockJam/Data/Environment/EnvironmentDatabase.asset";

        private static LevelData CreateSeed()
        {
            Type window = Type.GetType("ColorBlockJam.Editor.Levels.LevelDatabaseWindow, ColorBlockJam.Editor", true);
            MethodInfo method = window.GetMethod("CreateStarterLevel", BindingFlags.Static | BindingFlags.NonPublic);
            Assert.That(method, Is.Not.Null);
            return (LevelData)method.Invoke(null, new object[] { "level-001", "Level 1" });
        }

        private static LevelData CreateOldLowercaseSeed()
        {
            return new LevelData(
                "level-001", "Level 1", 3, 3, 60,
                new[] { new LevelDoorData("d1", BoardSide.Top, 1, 1, "red") },
                new[] { new LevelBlockData("b1", "red", "Single", new Vector2Int(1, 1), new[] { Vector2Int.zero }) });
        }

        [Test]
        public void CreateLevelSeedShowsInThePreviewHostWithoutError()
        {
            BlockShapeLibrary library = AssetDatabase.LoadAssetAtPath<BlockShapeLibrary>(BoardKitAssets.ShapesPath);
            Scene scene = SceneManager.GetActiveScene();
            LevelPreviewHost host = new LevelPreviewHost(scene,
                AssetDatabase.LoadAssetAtPath<GameObject>(k_prefabPath),
                AssetDatabase.LoadAssetAtPath<EnvironmentDatabase>(k_databasePath),
                library);
            try
            {
                bool shown = host.TryShow(CreateSeed(), out string error);
                Debug.Log("VERIFIER-PROBE seed TryShow=" + shown + " error='" + error + "' blocks=" + host.ActiveBlocks.Count
                    + " levelId=" + host.LevelId);
                Assert.That(shown, Is.True, error);
                Assert.That(host.ActiveBlocks.Count, Is.EqualTo(1));
                Assert.That(host.LevelId, Is.EqualTo("level-001"));

                bool oldShown = host.TryShow(CreateOldLowercaseSeed(), out string oldError);
                Debug.Log("VERIFIER-PROBE old-lowercase TryShow=" + oldShown + " error='" + oldError + "'");
                Assert.That(oldShown, Is.False, "Negative control: the pre-fix lowercase seed must be rejected.");
            }
            finally
            {
                host.Dispose();
            }
        }
    }
}
