import json, copy
def base():
    return {"id":"vf","name":"Verify","board":{"width":5,"height":5,"voidCells":[]},"timerSeconds":120,
      "doors":[{"id":"d1","side":"Top","start":1,"length":2,"colorId":"Red","effects":[]},
               {"id":"d2","side":"Bottom","start":3,"length":1,"colorId":"Blue","effects":[]}],
      "blocks":[{"id":"b1","colorId":"Red","shapeId":"DoubleVertical","origin":{"x":0,"y":0},"cells":[{"x":0,"y":0},{"x":0,"y":1}],"effects":[]},
                {"id":"b2","colorId":"Blue","shapeId":"DoubleVertical","origin":{"x":3,"y":0},"cells":[{"x":0,"y":0},{"x":0,"y":1}],"effects":[]}],
      "cellObjects":[]}
def eff(t, *kv): return {"type":t,"params":[{"key":k,"value":v} for k,v in kv]}
def obj(i, cells, e): return {"id":i,"cells":[{"x":x,"y":y} for x,y in cells],"effect":e}
V={}
def add(name, mut, version=2):
    l=base(); mut(l); V[name]={"formatVersion":version,"levels":[l]}
add("valid_plain", lambda l: None)
add("valid_arrow", lambda l: l["blocks"][0]["effects"].append(eff("arrow",("axis","Vertical"))))
add("valid_barrier_open", lambda l: l["cellObjects"].append(obj("o1",[(0,0),(0,1)],eff("barrier",("startsOpen","true")))))

add("valid_void", lambda l: l["board"].__setitem__("voidCells",[{"x":4,"y":0}]))

add("inv_wrong_target", lambda l: l["doors"][0]["effects"].append(eff("arrow",("axis","Vertical"))))
add("inv_same_type_twice", lambda l: l["blocks"][0]["effects"].extend([eff("arrow",("axis","Vertical")),eff("arrow",("axis","Vertical"))]))
add("inv_unknown_key", lambda l: l["blocks"][0]["effects"].append(eff("arrow",("axis","Vertical"),("speed","2"))))

add("inv_missing_reference", lambda l: l["blocks"][0]["effects"].append(eff("combined",("partner","b9"))))
add("inv_exclusive_set", lambda l: l["blocks"][0]["effects"].extend([eff("moving-obstacle"),eff("star")]))
add("inv_non_string_value", lambda l: l["blocks"][0]["effects"].append({"type":"arrow","params":[{"key":"axis","value":1}]}))
add("inv_void_outside", lambda l: l["board"].__setitem__("voidCells",[{"x":5,"y":0}]))
add("inv_void_duplicate", lambda l: l["board"].__setitem__("voidCells",[{"x":4,"y":0},{"x":4,"y":0}]))

add("inv_obj_on_void", objvoid)
