// Barrier pieces over the given cells: count, height range, rendered count and per-quadrant coverage.
System.Func<UnityEngine.Vector2Int[], string> inspect = (cells) => {
  var root = UnityEngine.GameObject.Find("Level Preview (not saved)");
  int n = 0, shown = 0; float min = 9, max = -9; var quad = new System.Collections.Generic.Dictionary<UnityEngine.Vector2, int>();
  foreach (var p in root.GetComponentsInChildren<ColorBlockJam.Presentation.Board.BoardPieceView>(false)) {
    var lp = p.transform.localPosition; var c = new UnityEngine.Vector2Int(UnityEngine.Mathf.RoundToInt(lp.x), UnityEngine.Mathf.RoundToInt(lp.z));
    if (!(p.name.StartsWith("Wall") || p.name.StartsWith("Corner_1")) || p.transform.parent.name != "BoardRoot" || System.Array.IndexOf(cells, c) < 0) continue;
    n++; if (p.Renderer.enabled) shown++; min = UnityEngine.Mathf.Min(min, lp.y); max = UnityEngine.Mathf.Max(max, lp.y);
    var q = new UnityEngine.Vector2(UnityEngine.Mathf.Round(lp.x * 4f) / 4f, UnityEngine.Mathf.Round(lp.z * 4f) / 4f);
    int k; quad.TryGetValue(q, out k); quad[q] = k + 1;
  }
  int good = 0; foreach (var cell in cells) for (int qi = 0; qi < 4; qi++) {
    var q = new UnityEngine.Vector2(cell.x - 0.25f + 0.5f * (qi & 1), cell.y - 0.25f + 0.5f * (qi >> 1)); int k; quad.TryGetValue(q, out k); if (k == 1) good++; }
  return n + " pieces y[" + min.ToString("0.###") + ".." + max.ToString("0.###") + "] rendered " + shown + " quadrantsWithExactlyOnePiece " + good + "/" + (cells.Length * 4);
};
